package com.enterprise.banking.domain.usecase

import javax.inject.Inject
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.flowOn
import kotlinx.coroutines.Dispatchers

/**
 * Pure Kotlin Use Case without Android Framework dependencies.
 * Guarantees domain logic testability with standard JUnit.
 */
class GetLoanApplicationDetailsUseCase @Inject constructor(
    private val repository: LoanRepository
) {
    operator fun invoke(applicationId: String): Flow<Resource<LoanApplication>> {
        return repository.getLoanDetails(applicationId)
            .flowOn(Dispatchers.IO)
    }
}
package com.enterprise.banking.di

import dagger.Module
import dagger.Provides
import dagger.hilt.InstallIn
import dagger.hilt.components.SingletonComponent
import javax.inject.Singleton

@Module
@InstallIn(SingletonComponent::class)
object NetworkModule {

    @Provides
    @Singleton
    fun provideOkHttpClient(authInterceptor: AuthInterceptor): OkHttpClient {
        return OkHttpClient.Builder()
            .addInterceptor(authInterceptor)
            .connectTimeout(30, TimeUnit.SECONDS)
            .build()
    }

    @Provides
    @Singleton
    fun provideLoanRepository(impl: LoanRepositoryImpl): LoanRepository = impl
}
package com.enterprise.banking.presentation.loan

@HiltViewModel
class LoanDetailViewModel @Inject constructor(
    private val getLoanUseCase: GetLoanApplicationDetailsUseCase
) : ViewModel() {

    private val _uiState = MutableStateFlow<LoanUiState>(LoanUiState.Loading)
    val uiState: StateFlow<LoanUiState> = _uiState.asStateFlow()

    fun fetchDetails(id: String) {
        viewModelScope.launch {
            getLoanUseCase(id).collect { resource ->
                _uiState.value = when (resource) {
                    is Resource.Success -> LoanUiState.Success(resource.data)
                    is Resource.Error   -> LoanUiState.Error(resource.message)
                    is Resource.Loading -> LoanUiState.Loading
                }
            }
        }
    }
}
